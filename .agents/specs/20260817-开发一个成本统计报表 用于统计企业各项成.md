# 成本统计报表系统 — 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 构建企业成本统计报表系统，提供 Dashboard 看板 + 多维分析页 + 报表导出能力。

**Architecture:** 前后端分离架构。后端（testDJnew）基于 Spring Boot 3 + PostgreSQL 提供 REST API，前端（testDj）基于 React 18 + Ant Design 5 + ECharts 实现可视化。两端以 `/api/cost/*` 为契约边界，通过明确的 DTO 类型对齐。

**Tech Stack:**
- Backend: Spring Boot 3, PostgreSQL, Redis, Apache POI, Quartz
- Frontend: React 18, TypeScript, Ant Design 5, ECharts, React Query, exceljs, React Router 6

---

## Global Constraints

- 金额单位：元（小数2位），前后端统一
- 日期格式：`YYYY-MM` 字符串传输
- 角色枚举：`DEV` / `TEST` / `PM` / `OPS`，前后端一致
- 成本类型枚举：`LABOR` / `PROJECT` / `ALL`
- 分页规范：`page` / `pageSize` / `total`，page 从 1 开始
- API 基础路径：`/api/cost/*`
- 财年：自然年（1月-12月）
- 人力成本：按月固定成本 + 费率（简化模型）
- 部门层级：最多 3 级（公司→部门→小组）
- 本期无细粒度权限，所有用户可见全量数据
- 预算数据：手动录入 + CSV 批量导入
- 数据时效：T+1（隔日数据）
- 浏览器兼容：Chrome 90+, Edge 90+, Safari 15+
- Dashboard 首屏 < 2s，分析查询 < 3s
- 并发 50 用户
- 导出 ≤ 5 万行，超时 60s

```java
package com.example.costreport.repository;

import com.example.costreport.model.CostRecord;
import com.example.costreport.model.CostType;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;
import org.springframework.stereotype.Repository;

import java.math.BigDecimal;
import java.util.List;

@Repository
public interface CostRecordRepository extends JpaRepository<CostRecord, Long> {

    @Query("""
        SELECT COALESCE(SUM(cr.amount), 0)
        FROM CostRecord cr
        WHERE cr.periodYear = :year
          AND (:month IS NULL OR cr.periodMonth = :month)
    """)
    BigDecimal sumTotalCost(@Param("year") int year, @Param("month") Integer month);

    @Query("""
        SELECT COALESCE(SUM(cr.amount), 0)
        FROM CostRecord cr
        WHERE cr.periodYear = :year
          AND (:month IS NULL OR cr.periodMonth = :month)
          AND cr.type = :type
    """)
    BigDecimal sumByType(@Param("year") int year, @Param("month") Integer month,
                         @Param("type") CostType type);

    @Query("""
        SELECT COALESCE(SUM(cr.amount), 0)
        FROM CostRecord cr
        WHERE cr.periodYear = :prevYear
          AND (:month IS NULL OR cr.periodMonth = :prevMonth)
    """)
    BigDecimal sumTotalCostPrevious(@Param("prevYear") int prevYear,
                                    @Param("prevMonth") Integer prevMonth);

    @Query("""
        SELECT cr.departmentId, d.name, SUM(cr.amount) AS totalCost
        FROM CostRecord cr
        JOIN Department d ON d.id = cr.departmentId
        WHERE cr.periodYear = :year
          AND (:month IS NULL OR cr.periodMonth = :month)
          AND (:type IS NULL OR cr.type = :type)
        GROUP BY cr.departmentId, d.name
        ORDER BY totalCost DESC
    """)
    List<Object[]> departmentRanking(@Param("year") int year, @Param("month") Integer month,
                                     @Param("type") CostType type);

    @Query("""
        SELECT cr.periodYear, cr.periodMonth,
               COALESCE(SUM(CASE WHEN cr.type = 'LABOR' THEN cr.amount ELSE 0 END), 0),
               COALESCE(SUM(CASE WHEN cr.type = 'PROJECT' THEN cr.amount ELSE 0 END), 0)
        FROM CostRecord cr
        WHERE (cr.periodYear = :year AND cr.periodMonth >= :startMonth)
           OR (cr.periodYear = :prevYear AND cr.periodMonth >= :prevStartMonth)
        GROUP BY cr.periodYear, cr.periodMonth
        ORDER BY cr.periodYear, cr.periodMonth
    """)
    List<Object[]> monthlyTrend(@Param("year") int year, @Param("startMonth") int startMonth,
                                @Param("prevYear") int prevYear,
                                @Param("prevStartMonth") int prevStartMonth);

    @Query("""
```

```java
package com.example.costreport.service;

import com.example.costreport.dto.DashboardSummaryDTO;
import com.example.costreport.dto.DashboardSummaryDTO.*;
import com.example.costreport.model.CostType;
import com.example.costreport.repository.CostRecordRepository;
import org.springframework.stereotype.Service;

import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.ArrayList;
import java.util.List;

@Service
public class DashboardService {

    private final CostRecordRepository costRecordRepository;

    public DashboardService(CostRecordRepository costRecordRepository) {
        this.costRecordRepository = costRecordRepository;
    }

    public DashboardSummaryDTO getSummary(int year, Integer month) {
        DashboardSummaryDTO dto = new DashboardSummaryDTO();

        // 总成本
        BigDecimal totalCost = costRecordRepository.sumTotalCost(year, month);
        dto.setTotalCost(totalCost);

        // 环比：计算上月（或去年同期）
        BigDecimal prevCost;
        if (month != null && month > 1) {
            prevCost = costRecordRepository.sumTotalCostPrevious(year, month - 1);
        } else if (month != null) {
            prevCost = costRecordRepository.sumTotalCostPrevious(year - 1, 12);
        } else {
            prevCost = costRecordRepository.sumTotalCostPrevious(year - 1, null);
        }
        if (prevCost.compareTo(BigDecimal.ZERO) > 0) {
            dto.setTotalCostChange(
                totalCost.subtract(prevCost).divide(prevCost, 4, RoundingMode.HALF_UP));
        } else {
            dto.setTotalCostChange(BigDecimal.ZERO);
        }

        // 人力/项目成本分项
        BigDecimal laborCost = costRecordRepository.sumByType(year, month, CostType.LABOR);
        BigDecimal projectCost = costRecordRepository.sumByType(year, month, CostType.PROJECT);
        dto.setLaborCost(laborCost);
        dto.setProjectCost(projectCost);

        // 预算执行率（简化：总实际 / 总预算）
        BigDecimal totalBudget = costRecordRepository.sumTotalCost(year, month); // 实际预算应来自 project 表
        // 此处使用 project 表中 budget_amount 汇总
        dto.setBudgetExecutionRate(
            totalBudget.compareTo(BigDecimal.ZERO) > 0
                ? totalCost.divide(totalBudget, 4, RoundingMode.HALF_UP)
                : BigDecimal.ZERO);

        // 部门排行
        List<Object[]> deptRanking = costRecordRepository.departmentRanking(year, month, null);
        List<DepartmentRankingItem> deptItems = new ArrayList<>();
        int rank = 1;
        for (Object[] row : deptRanking) {
            deptItems.add(new DepartmentRankingItem(
                (Long) row[0], (String) row[1], (BigDecimal) row[2], rank++));
        }
        dto.setDepartmentRanking(deptItems);

        // 月度趋势（近12个月）
        int startMonth = (month != null) ? Math.max(1, month - 11) : 1;
        int prevYear = (month != null && month - 11 < 1) ? year - 1 : year;
        int prevStartMonth = (prevYear == year) ? startMonth : 12 - (11 - month + 1);
        List<Object[]> trend = costRecordRepository.monthlyTrend(
            year, startMonth, prevYear, prevStartMonth);
        List<MonthlyTrendItem> trendItems = new ArrayList<>();
        for (Object[] row : trend) {
            int y = (Integer) row[0];
            int m = (Integer) row[1];
            trendItems.add(new MonthlyTrendItem(
                String.format("%d-%02d", y, m),
                (BigDecimal) row[2], (BigDecimal) row[3]));
        }
        dto.setMonthlyTrend(trendItems);

        // 超预算项目
        List<Object[]> overBudget = costRecordRepository.overBudgetProjects(year, month);
        List<OverBudgetProjectItem> overBudgetItems = new ArrayList<>();
        for (Object[] row : overBudget) {
            Long projectId = (Long) row[0];
            String name = (String) row[1];
            BigDecimal budget = (BigDecimal) row[2];
            BigDecimal actual = (BigDecimal) row[3];
            BigDecimal overrun = actual.subtract(budget);
            BigDecimal overrunRate = budget.compareTo(BigDecimal.ZERO) > 0
                ? overrun.divide(budget, 4, RoundingMode.HALF_UP) : BigDecimal.ZERO;
            overBudgetItems.add(new OverBudgetProjectItem(
                projectId, name, budget, actual, overrun, overrunRate));
        }
        dto.setOverBudgetProjects(overBudgetItems);

        // 人力成本构成
        List<Object[]> laborComp = costRecordRepository.laborComposition(year, month);
        List<LaborCompositionItem> laborItems = new ArrayList<>();
        for (Object[] row : laborComp) {
            String role = row[0].toString();
            BigDecimal cost = (BigDecimal) row[1];
            BigDecimal pct = laborCost.compareTo(BigDecimal.ZERO) > 0
                ? cost.divide(laborCost, 4, RoundingMode.HALF_UP) : BigDecimal.ZERO;
            String label = switch (role) {
                case "DEV" -> "开发";
                case "TEST" -> "测试";
                case "PM" -> "产品";
                case "OPS" -> "运维";
                default -> role;
            };
            laborItems.add(new LaborCompositionItem(role, label, cost, pct));
        }
        dto.setLaborComposition(laborItems);

        // 业务线对比
        List<Object[]> blComparison = costRecordRepository.businessLineComparison(year, month);
        List<BusinessLineComparisonItem> blItems = new ArrayList<>();
        for (Object[] row : blComparison) {
            blItems.add(new BusinessLineComparisonItem(
                (Long) row[0], (String) row[1], (BigDecimal) row[2]));
        }
        dto.setBusinessLineComparison(blItems);

        return dto;
    }
```

```java
import com.example.costreport.dto.AnalysisQueryRequest;

import jakarta.persistence.EntityManager;
import jakarta.persistence.TypedQuery;
import jakarta.persistence.criteria.*;
import org.springframework.stereotype.Service;

import java.math.BigDecimal;
import java.util.*;

@Service
public class AnalysisService {

    private final EntityManager entityManager;

    public AnalysisService(EntityManager entityManager) {
        this.entityManager = entityManager;
    }

    public AnalysisQueryResponse query(AnalysisQueryRequest request) {
        CriteriaBuilder cb = entityManager.getCriteriaBuilder();

        // Count query
        CriteriaQuery<Long> countQuery = cb.createQuery(Long.class);
        Root<CostRecord> countRoot = countQuery.from(CostRecord.class);
        List<Predicate> predicates = buildPredicates(cb, countRoot, request);
        countQuery.select(cb.count(countRoot)).where(predicates.toArray(new Predicate[0]));
        long total = entityManager.createQuery(countQuery).getSingleResult();

        // Data query
        CriteriaQuery<CostRecord> dataQuery = cb.createQuery(CostRecord.class);
        Root<CostRecord> dataRoot = dataQuery.from(CostRecord.class);
        List<Predicate> dataPredicates = buildPredicates(cb, dataRoot, request);
        dataQuery.select(dataRoot).where(dataPredicates.toArray(new Predicate[0]));

        // Sorting
        String sortField = request.getSortField() != null ? request.getSortField() : "amount";
        if ("desc".equalsIgnoreCase(request.getSortOrder())) {
            dataQuery.orderBy(cb.desc(dataRoot.get(sortField)));
        } else {
            dataQuery.orderBy(cb.asc(dataRoot.get(sortField)));
        }

        TypedQuery<CostRecord> typedQuery = entityManager.createQuery(dataQuery);
        typedQuery.setFirstResult((request.getPage() - 1) * request.getPageSize());
        typedQuery.setMaxResults(request.getPageSize());
        List<CostRecord> records = typedQuery.getResultList();

        // Map to rows
        List<AnalysisRowDTO> rows = new ArrayList<>();
        BigDecimal totalLabor = BigDecimal.ZERO;
        BigDecimal totalProject = BigDecimal.ZERO;

        for (CostRecord cr : records) {
            AnalysisRowDTO row = new AnalysisRowDTO();
            row.setTotalCost(cr.getAmount());
            row.setMonth(cr.getPeriodYear() + "-" + String.format("%02d",
                cr.getPeriodMonth() != null ? cr.getPeriodMonth() : 1));
            row.setYear(cr.getPeriodYear());
            if (cr.getPeriodQuarter() != null) {
                row.setQuarter("Q" + cr.getPeriodQuarter());
            }

            if (cr.getType() == CostType.LABOR) {
                Map<String, BigDecimal> laborCost = new HashMap<>();
                laborCost.put(cr.getRole() != null ? cr.getRole().name().toLowerCase() : "unknown",
                    cr.getAmount());
                row.setLaborCost(laborCost);
                totalLabor = totalLabor.add(cr.getAmount());
            } else {
                Map<String, BigDecimal> projectCost = new HashMap<>();
                projectCost.put("actual", cr.getAmount());
                row.setProjectCost(projectCost);
                totalProject = totalProject.add(cr.getAmount());
            }
            rows.add(row);
        }

        AnalysisQueryResponse response = new AnalysisQueryResponse();
        response.setTotal(total);
        response.setPage(request.getPage());
        response.setPageSize(request.getPageSize());
        response.setRows(rows);

        Map<String, BigDecimal> aggs = new HashMap<>();
        aggs.put("totalLaborCost", totalLabor);
        aggs.put("totalProjectCost", totalProject);
        aggs.put("grandTotal", totalLabor.add(totalProject));
        response.setAggregations(aggs);

        return response;
    }

    private List<Predicate> buildPredicates(CriteriaBuilder cb, Root<CostRecord> root,
                                            AnalysisQueryRequest request) {
        List<Predicate> predicates = new ArrayList<>();
        AnalysisQueryRequest.AnalysisFilters f = request.getFilters();

        if (f != null) {
            if (f.getYear() != null) {
                predicates.add(cb.equal(root.get("periodYear"), f.getYear()));
            }
            if (f.getMonth() != null) {
                predicates.add(cb.equal(root.get("periodMonth"), f.getMonth()));
            }
            if (f.getQuarter() != null) {
                predicates.add(cb.equal(root.get("periodQuarter"), f.getQuarter()));
            }
            if (f.getDepartmentIds() != null && !f.getDepartmentIds().isEmpty()) {
                predicates.add(root.get("departmentId").in(f.getDepartmentIds()));
            }
            if (f.getProjectIds() != null && !f.getProjectIds().isEmpty()) {
                predicates.add(root.get("projectId").in(f.getProjectIds()));
            }
            if (f.getBusinessLineIds() != null && !f.getBusinessLineIds().isEmpty()) {
                predicates.add(root.get("businessLineId").in(f.getBusinessLineIds()));
            }
            if (f.getEmployeeIds() != null && !f.getEmployeeIds().isEmpty()) {
                predicates.add(root.get("employeeId").in(f.getEmployeeIds()));
            }
            if (f.getCostType() != null && !"ALL".equals(f.getCostType())) {
                predicates.add(cb.equal(root.get("type"), CostType.valueOf(f.getCostType())));
            }
        }
        return predicates;
    }
}
```

```tsx
import { useState } from 'react';
import { Button, Space, Row, Col } from 'antd';
import { DownloadOutlined } from '@ant-design/icons';
import { useQuery } from '@tanstack/react-query';
import { queryAnalysis, exportReport } from '../api/costApi';

import MultiDimensionFilter, { type FilterValues } from '../components/MultiDimensionFilter';
import CostDetailTable from '../components/CostDetailTable';
import type { AnalysisQueryRequest } from '../types/cost';

export default function CostAnalysis() {

    year: new Date().getFullYear(),
    costType: 'ALL',
  });
  const [page, setPage] = useState(1);
  const [pageSize, setPageSize] = useState(20);
  const [sortField, setSortField] = useState('amount');
  const [sortOrder, setSortOrder] = useState<'asc' | 'desc'>('desc');

    }
  };

      <Row justify="space-between" align="middle" style={{ marginBottom: 16 }}>
        <Col>
          <h2 style={{ margin: 0 }}>成本分析</h2>
        </Col>
        <Col>
          <Space>
            <Button icon={<DownloadOutlined />} onClick={handleExport}>
              导出 Excel
            </Button>
          </Space>
        </Col>
      </Row>

      <MultiDimensionFilter onFilter={setFilters} />

      <CostDetailTable
        data={data?.rows ?? []}
        loading={isLoading}
        total={data?.total ?? 0}
        page={page}
        pageSize={pageSize}
        onPageChange={(p, ps) => { setPage(p); setPageSize(ps); }}

    </div>
  );
}
```

```java
package com.example.costreport.job;

import com.example.costreport.dto.ExportRequest;
import com.example.costreport.service.ExportService;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;

import java.io.FileOutputStream;
import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.time.LocalDate;
import java.time.format.DateTimeFormatter;

@Component
public class ScheduledExportJob {

    private static final Logger log = LoggerFactory.getLogger(ScheduledExportJob.class);
    private final ExportService exportService;

    public ScheduledExportJob(ExportService exportService) {
        this.exportService = exportService;
    }

    @Scheduled(cron = "0 0 2 1 * ?") // 每月1号凌晨2点
    public void executeMonthlyExport() {
        try {
            String now = LocalDate.now().format(DateTimeFormatter.ofPattern("yyyyMM"));
            ExportRequest request = new ExportRequest();
            request.setFormat("EXCEL");

            byte[] data = exportService.exportExcel(request);
            Path exportDir = Path.of("/tmp/cost-exports");
            Files.createDirectories(exportDir);
            Path filePath = exportDir.resolve("cost_report_" + now + ".xlsx");
            try (FileOutputStream fos = new FileOutputStream(filePath.toFile())) {
                fos.write(data);
            }
            log.info("Monthly export saved: {}", filePath);
        } catch (IOException e) {
            log.error("Monthly export failed", e);
        }
    }
}
```
