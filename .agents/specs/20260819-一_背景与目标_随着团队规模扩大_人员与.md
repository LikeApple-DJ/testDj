## 1. 通览 (Overview)

本计划基于 `dima.md` 需求规格说明书，将**组织架构管理模块**拆解为可独立交付、可测试的 8 个任务。

**核心交付物:**
- 部门树形结构管理（懒加载、拖拽、防循环引用）
- 员工全生命周期管理（新增/调动/离职）
- 审批流级联更新 + 调动历史留痕
- 乐观锁并发控制 + 逻辑删除

**技术栈假设:** 后端 Spring Boot 3.x + MyBatis-Plus + MySQL 8，前端 React 18 + Ant Design 5 + Redux Toolkit。

CREATE TABLE IF NOT EXISTS department (
    id          BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '主键',
    name        VARCHAR(100) NOT NULL COMMENT '部门名称',
    parent_id   BIGINT DEFAULT NULL COMMENT '父部门ID',

    updated_at    DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

@Override
public List<DepartmentTreeDTO> getDepartmentTree() {
    List<Department> all = list();
    return buildTree(all, null);
}

private List<DepartmentTreeDTO> buildTree(List<Department> all, Long parentId) {
    return all.stream()
        .filter(d -> (parentId == null && d.getParentId() == null)
                     || (parentId != null && parentId.equals(d.getParentId())))
        .map(d -> {
            DepartmentTreeDTO dto = new DepartmentTreeDTO();
            dto.setId(d.getId());
            dto.setName(d.getName());
            dto.setChildren(buildTree(all, d.getId()));
            return dto;
        })
        .toList();
}

@Override
public void moveDepartment(Long id, Long newParentId) {
    if (newParentId != null && isDescendant(newParentId, id)) {
        throw new BusinessException("不能将部门移动到自己的子孙节点下");
    }
    Department dept = getById(id);
    dept.setParentId(newParentId);
    // 重新计算 path
    String parentPath = newParentId == null ? "" : getById(newParentId).getPath();
    dept.setPath(parentPath + "-" + id);
    updateById(dept);
}

private boolean isDescendant(Long ancestorId, Long targetId) {
    List<Department> children = lambdaQuery().eq(Department::getParentId, ancestorId).list();
    for (Department child : children) {
        if (child.getId().equals(targetId) || isDescendant(child.getId(), targetId)) {
            return true;
        }
    }
    return false;
}
```

@Override
@Transactional(rollbackFor = Exception.class)
public void transferEmployee(Long id, TransferDTO dto) {
    Employee employee = getById(id);
    if (employee == null) {
        throw new BusinessException("员工不存在");
    }
    Long oldDeptId = employee.getDeptId();
    String oldPosition = employee.getPosition();

    // 更新员工部门/职位（乐观锁由 MP 自动处理 version 字段）
    employee.setDeptId(dto.getNewDeptId());
    employee.setPosition(dto.getNewPosition());
    boolean updated = updateById(employee);
    if (!updated) {
        throw new BusinessException("该员工信息已被他人修改，请刷新重试");
    }

    // 写入调动记录
    TransferRecord record = new TransferRecord();
    record.setEmployeeId(id);
    record.setOldDeptId(oldDeptId);
    record.setNewDeptId(dto.getNewDeptId());
    record.setOldPosition(oldPosition);
    record.setNewPosition(dto.getNewPosition());
    record.setReason(dto.getReason());
    record.setOperatorId(getCurrentUserId());
    transferRecordMapper.insert(record);

    // 触发审批流级联更新（事件驱动，解耦）
    eventPublisher.publishEvent(new EmployeeTransferredEvent(id, oldDeptId, dto.getNewDeptId()));
}
```

> **注意**: `updateById` 在 MyBatis-Plus 乐观锁插件启用时，若 `version` 不匹配则返回 `false`。此时应抛出自定义异常，Controller 层捕获后返回 `409 Conflict`。

@Override
@Transactional(rollbackFor = Exception.class)
public void resignEmployee(Long id, ResignDTO dto) {
    Employee employee = getById(id);
    if (employee == null || employee.getStatus() == 0) {
        throw new BusinessException("员工不存在或已离职");
    }
    employee.setStatus(0); // 0=离职
    employee.setIsDeleted(1); // 逻辑删除
    updateById(employee);

    // 释放登录权限（调用下游 AuthService）
    authService.revokeLoginPermission(id);
}

public Result<Void> delete(@PathVariable Long id) {

        .eq(Employee::getDeptId, id)
        .eq(Employee::getIsDeleted, 0)
        .eq(Employee::getStatus, 1)
